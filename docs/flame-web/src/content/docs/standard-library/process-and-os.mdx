## Child Processes (`std.process`)

Execute terminal commands and capture output or exit codes:

```flame
import std.process

fn main() {
    // Run a command and capture output
    let output = process.exec("git", ["status", "--short"])
    print($"Git status:\n{output.stdout}")

    // Run command and check exit code
    let status = process.run("cargo", ["--version"])
    print($"Exit code: {status.code}")

    // Exit the current process
    if status.code != 0 {
        process.exit(1)
    }
}
```

## Environment Variables (`std.env`)

```flame
import std.env

fn main() {
    // Read environment variable
    let port = env.get("PORT")
    if port == Nil {
        port = "8080"
    }

    // Set environment variable for child processes
    env.set("APP_ENV", "production")

    // Get current working directory
    let current_dir = env.current_dir()
    print($"CWD: {current_dir}")
}
```

## OS Introspection (`std.os`)

```flame
import std.os

fn main() {
    print($"OS Family: {os.family()}") // "windows", "linux", or "macos"
    print($"CPU Architecture: {os.arch()}") // "x86_64", "aarch64"
    print($"Total Logical Cores: {os.cpu_count()}")
}
```
